  function addMember(e) {
    e.preventDefault();
    const nameEl = document.getElementById('newMemberName');
    const upiEl = document.getElementById('newMemberUpi');
    const name = nameEl.value.trim();
    const upi = upiEl.value.trim();
    if (!name || !upi) return;

    members.push({ id: 'm' + Date.now(), name, upi });
    nameEl.value = '';
    upiEl.value = '';
    saveState();
    renderAll();
  }

  function removeMember(id) {
    if (members.length <= 2) {
      alert('Minimum 2 roommates are required.');
      return;
    }
    const involved = expenses.some(ex => ex.paidBy === id || (ex.splits[id] && ex.splits[id] > 0));
    if (involved) {
      alert('Cannot remove a member who is already part of an existing expense.');
      return;
    }
    members = members.filter(m => m.id !== id);
    saveState();
    renderAll();
  }

  function setSplitMode(mode) {
    currentMode = mode;
    ['equal', 'unequal', 'shares', 'itemwise'].forEach(m => {
      const btn = document.getElementById('tab-' + m);
      if (m === mode) {
        btn.classList.add('tab-active');
        btn.classList.remove('text-gray-600');
      } else {
        btn.classList.remove('tab-active');
        btn.classList.add('text-gray-600');
      }
    });

    const totalWrapper = document.getElementById('totalAmountWrapper');
    if (mode === 'itemwise') {
      totalWrapper.classList.add('hidden');
    } else {
      totalWrapper.classList.remove('hidden');
    }

    document.getElementById('splitError').classList.add('hidden');
    renderSplitPanel();
  }

  function handleBillUpload(e) {
    const file = e.target.files[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = function(ev) {
      document.getElementById('billPreviewImg').src = ev.target.result;
      document.getElementById('billPreviewBox').classList.remove('hidden');
      if (!document.getElementById('expTitle').value) {
        document.getElementById('expTitle').value = 'Receipt: ' + file.name.replace(/\.[^/.]+$/, "");
      }
    };
    reader.readAsDataURL(file);
  }

  function clearBillPhoto() {
    document.getElementById('billPreviewBox').classList.add('hidden');
    document.getElementById('billPreviewImg').src = '';
  }

  function renderBillItems() {
    const box = document.getElementById('itemsContainer');
    if (!box) return;
    let subTotal = 0;

    box.innerHTML = billItems.map((item, idx) => {
      subTotal += item.price;
      return `
        <div class="bg-white border border-gray-200 rounded-lg p-2.5">
          <div class="flex justify-between items-center mb-1.5">
            <span class="text-xs font-semibold text-gray-800">${item.name} — <span class="text-emerald-700">₹${item.price}</span></span>
            <button type="button" onclick="removeBillItem(${idx})" class="text-xs text-gray-400 hover:text-red-500">&times;</button>
          </div>
          <div class="flex flex-wrap gap-1.5">
            ${members.map(m => {
              const active = item.sharedBy.includes(m.id);
              return `
                <button type="button" onclick="toggleItemMember(${idx}, '${m.id}')"
                  class="text-[11px] px-2 py-0.5 rounded-full border transition ${active ? 'bg-emerald-600 text-white border-emerald-600' : 'bg-gray-50 text-gray-600 border-gray-300'}">
                  ${m.name}
                </button>
              `;
            }).join('')}
          </div>
        </div>
      `;
    }).join('');

    const grand = subTotal + billTax;
    const gtEl = document.getElementById('itemWiseGrandTotal');
    if (gtEl) gtEl.textContent = `Grand Total (Items ₹${subTotal} + Tax ₹${billTax}): ₹${grand.toFixed(2)}`;
  }

  function addBillItem() {
    const nameEl = document.getElementById('newItemName');
    const priceEl = document.getElementById('newItemPrice');
    const name = nameEl.value.trim();
    const price = parseFloat(priceEl.value);
    if (!name || isNaN(price) || price <= 0) return;

    billItems.push({
      id: Date.now(),
      name,
      price,
      sharedBy: members.map(m => m.id)
    });
    nameEl.value = '';
    priceEl.value = '';
    renderBillItems();
  }

  function removeBillItem(idx) {
    billItems.splice(idx, 1);
    renderBillItems();
  }
